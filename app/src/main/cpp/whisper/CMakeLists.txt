cmake_minimum_required(VERSION 3.22)
project(onhand_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN YES)
if(NOT IS_DIRECTORY "${ONHAND_WHISPER_SOURCE}")
    message(FATAL_ERROR "Set ONHAND_WHISPER_SOURCE to the pinned whisper.cpp source")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_WEBGPU OFF CACHE BOOL "" FORCE)
set(GGML_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "OnHand Whisper supports arm64-v8a")
    endif()
    set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
endif()
add_subdirectory("${ONHAND_WHISPER_SOURCE}" whisper-upstream EXCLUDE_FROM_ALL)
add_library(onhand_whisper_core STATIC whisper_core.cpp)
target_link_libraries(onhand_whisper_core PUBLIC whisper)
target_include_directories(onhand_whisper_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

option(ONHAND_WHISPER_HOST_TEST "Build the real-model host smoke test" OFF)
if(ONHAND_WHISPER_HOST_TEST)
    add_executable(onhand_whisper_smoke whisper_smoke.cpp)
    target_link_libraries(onhand_whisper_smoke PRIVATE onhand_whisper_core)
endif()
if(ANDROID OR ONHAND_WHISPER_HOST_JNI)
    if(NOT ANDROID)
        find_package(JNI REQUIRED)
    endif()
    add_library(onhand_whisper SHARED whisper_jni.cpp)
    target_include_directories(onhand_whisper PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(onhand_whisper PRIVATE onhand_whisper_core)
    target_link_options(onhand_whisper PRIVATE
        "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
        "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic" "-Wl,-z,max-page-size=16384")
endif()
